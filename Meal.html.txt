/* ---------------- Datum ---------------- */

function fmtISO(d) {
  return `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')}`;
}
function addDays(iso, n) {
  const d = new Date(iso + 'T00:00:00');
  d.setDate(d.getDate() + n);
  return fmtISO(d);
}
function mondayOf(iso) {
  const d = new Date(iso + 'T00:00:00');
  const day = (d.getDay() + 6) % 7;
  d.setDate(d.getDate() - day);
  return fmtISO(d);
}
function fmtDE(iso) {
  const [y,m,d] = iso.split('-');
  return `${d}.${m}.${y}`;
}
function getTodayISO() { return fmtISO(new Date()); }

/* ---------------- API: DeepSeek + Anthropic ---------------- */

async function callAI(prompt, maxTokens = 4096, retries = 3) {
  const apiKey = state.settings.apiKey;
  if (!apiKey) {
    toast('⚠️ Bitte API-Key in den Einstellungen hinterlegen');
    throw new Error('Kein API-Key');
  }

  const provider = state.settings.provider || 'deepseek';
  const tokens = Math.min(maxTokens, state.settings.maxTokens || 4096);
  const temperature = state.settings.temperature || 0.7;

  let endpoint, headers, body;

  if (provider === 'deepseek') {
    endpoint = 'https://api.deepseek.com/v1/chat/completions';
    headers = {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${apiKey}`
    };
    body = {
      model: 'deepseek-chat',
      messages: [{ role: 'user', content: prompt }],
      max_tokens: tokens,
      temperature: temperature,
      top_p: 0.95
    };
  } else {
    // Anthropic
    endpoint = 'https://api.anthropic.com/v1/messages';
    headers = {
      'Content-Type': 'application/json',
      'x-api-key': apiKey,
      'anthropic-version': '2023-06-01'
    };
    body = {
      model: 'claude-3-5-sonnet-20241022',
      max_tokens: tokens,
      temperature: temperature,
      top_p: 0.95,
      messages: [{ role: 'user', content: prompt }]
    };
  }

  for (let attempt = 1; attempt <= retries; attempt++) {
    try {
      const resp = await fetch(endpoint, {
        method: 'POST',
        headers: headers,
        body: JSON.stringify(body)
      });

      if (!resp.ok) {
        const errorData = await resp.json();
        throw new Error(`API-Fehler ${resp.status}: ${errorData.error?.message || errorData.error || resp.statusText}`);
      }

      const data = await resp.json();
      let text;

      if (provider === 'deepseek') {
        text = data.choices?.[0]?.message?.content || '';
      } else {
        text = (data.content || []).map(b => b.text || '').join('');
      }

      // JSON extrahieren
      const cleaned = text.replace(/^```json\s*|\s*```$/g, '').trim();
      const match = cleaned.match(/[\{\[][\s\S]*[\}\]]/);
      const jsonStr = match ? match[0] : cleaned;

      try {
        return JSON.parse(jsonStr);
      } catch (e) {
        const repaired = repairJSON(jsonStr);
        if (repaired) return repaired;
        throw new Error('Ungültiges JSON-Format');
      }

    } catch (error) {
      console.error(`Versuch ${attempt}/${retries} fehlgeschlagen:`, error);
      if (attempt === retries) {
        toast(`❌ KI-Fehler: ${error.message}`);
        throw error;
      }
      await new Promise(resolve => setTimeout(resolve, 1000 * attempt));
    }
  }
}

function repairJSON(jsonStr) {
  try {
    let fixed = jsonStr
      .replace(/,\s*}/g, '}')
      .replace(/,\s*\]/g, ']')
      .replace(/(['"])?([a-zA-Z0-9_]+)(['"])?:/g, '"$2":')
      .replace(/:(\s*)'([^']*)'/g, ':"$2"')
      .replace(/:\s*undefined/g, ':null')
      .replace(/:\s*NaN/g, ':null')
      .replace(/:\s*Infinity/g, ':null');
    return JSON.parse(fixed);
  } catch { return null; }
}

  return callAI(prompt, maxTokens, retries);
}

/* ---------------- Automatische Kategorisierung ---------------- */

function autoCategorize(ingredientName) {
  const name = ingredientName.toLowerCase().trim();
  for (const [category, keywords] of Object.entries(CATEGORY_KEYWORDS)) {
    for (const keyword of keywords) {
      if (name === keyword || name.startsWith(keyword + ' ') || name.endsWith(' ' + keyword) || name.includes(keyword)) {
        return category;
      }
    }
  }
  return "other";
}

async function autoCategorizeWithAI(ingredients) {
  if (!ingredients || ingredients.length === 0) return {};
  try {
    const prompt = `
Kategorisiere folgende Zutaten in diese Kategorien:
${Object.entries(CATEGORY_LABELS).map(([key, label]) => `- ${key}: ${label}`).join('\n')}

Zutaten:
${ingredients.map(ing => `- ${ing.name}`).join('\n')}

Antworte NUR als JSON-Objekt:
{
  "zutat1": "kategorie",
  "zutat2": "kategorie"
}`;
    return await callAI(prompt, 1024);
  } catch {
    const result = {};
    ingredients.forEach(ing => {
      result[ing.name] = autoCategorize(ing.name);
    });
    return result;
  }
}

async function generateRecipeWithAI() {
  try {
    toast('🌿 KI generiert Rezept...');
    const prompt = `
Erstelle ein kreatives Rezept mit:
- Name: Einprägsamer deutscher Name
- Portionen: 2-4
- Zutaten: 5-8 frische, saisonale Zutaten
- Kategorien: produce, dairy, meat, pantry, spices
- Preise: Realistische Preise in € (0.50-8.00)
- Anleitung: Klare, strukturierte Schritte (mit \\n)

Antworte NUR als JSON:
{
  "name": "Rezeptname",
  "servings": 2,
  "prepMinutes": 20,
  "calories": 450,
  "tags": "vegetarisch, schnell",
  "ingredients": [

  ],
  "instructions": "Schritt 1: ...\\nSchritt 2: ...",
  "equipment": ["Backofen", "Pfanne"],

}`;

    const result = await callAI(prompt, 4096);
    if (!result.name || !result.ingredients) {

      ingredients: result.ingredients.map(ing => ({

        category: categorized[ing.name] || autoCategorize(ing.name),

      }))
    };

    state.recipeDraft = recipe;
    state.editingRecipeId = null;
    toast('✅ Rezept von KI generiert! Prüfe und speichere es.');
    render();

  } catch (error) {

    toast(`❌ Fehler: ${error.message}`);
  }
}

async function estimatePrices() {
  const rows = document.querySelectorAll('#ingredient-rows .ing-row');
  const ingredients = [];
  rows.forEach(row => {
    const name = row.querySelector('.ing-name').value.trim();
    if (name) {
      ingredients.push({
        name,
        amount: parseFloat(row.querySelector('.ing-amount').value) || 1,
        unit: row.querySelector('.ing-unit').value.trim() || 'Stück',
        category: row.querySelector('.ing-category').value
      });
    }
  });

  if (ingredients.length === 0) {
    toast('Bitte füge zuerst Zutaten hinzu');
    return;
  }

  try {
    toast('💶 Schätze Preise...');
    const prompt = `
Gib realistische Preise in € für deutsche Supermärkte (Rewe/Edeka):
${ingredients.map(ing => `- ${ing.name} (${ing.amount} ${ing.unit})`).join('\n')}

Antworte NUR als JSON-Array:
[
  {"name": "Zutat", "price": 1.99}
]`;

    const result = await callAI(prompt, 2048);
    if (!Array.isArray(result)) throw new Error('Ungültiges Format');

    const priceMap = {};
    result.forEach(item => {
      if (item.name && item.price !== undefined) {
        priceMap[item.name.toLowerCase().trim()] = item.price;
      }
    });

    rows.forEach(row => {
      const name = row.querySelector('.ing-name').value.trim().toLowerCase();
      const priceInput = row.querySelector('.ing-price');
      if (priceMap[name]) {
        priceInput.value = priceMap[name].toFixed(2);
      }
    });

    toast('✅ Preise geschätzt! Bitte prüfe die Werte.');
  } catch (error) {
    console.error('Preis-Schätzung fehlgeschlagen:', error);
    toast(`❌ Fehler: ${error.message}`);
  }
}

async function categorizeIngredients() {
  const rows = document.querySelectorAll('#ingredient-rows .ing-row');
  const ingredients = [];

    rows.forEach(row => {
      const name = row.querySelector('.ing-name').value.trim();
      const select = row.querySelector('.ing-category');

      if (category && CATEGORIES.includes(category)) select.value = category;

  }

function saveRecipeForm() {
  const name = document.getElementById('f-name').value.trim();
  if (!name) { toast('Bitte einen Namen eingeben.'); return; }
  const equipment = Array.from(document.querySelectorAll('input[name="f-equipment"]:checked')).map(el => el.value);
  const rows = document.querySelectorAll('#ingredient-rows .ing-row');
  const ingredients = [];
  rows.forEach(row => {
    const ingName = row.querySelector('.ing-name').value.trim();
    if (!ingName) return;
    const amountRaw = row.querySelector('.ing-amount').value;
    const priceRaw = row.querySelector('.ing-price').value;
    ingredients.push({
      name: ingName,
      amount: amountRaw ? parseFloat(amountRaw) : null,
      unit: row.querySelector('.ing-unit').value.trim(),
      category: row.querySelector('.ing-category').value,
      price: priceRaw ? parseFloat(priceRaw) : null,
    });
  });
  if (ingredients.length === 0) { toast('Bitte mindestens eine Zutat angeben.'); return; }

  const recipeData = {
    id: state.editingRecipeId || uid(),
    name,
    servings: parseInt(document.getElementById('f-servings').value) || 2,
    prepMinutes: parseInt(document.getElementById('f-prep').value) || 20,
    calories: parseInt(document.getElementById('f-calories').value) || null,
    tags: document.getElementById('f-tags').value.trim(),
    sourceUrl: document.getElementById('f-source').value.trim(),
    ingredients,
    instructions: document.getElementById('f-instructions').value.trim(),
    equipment,
  };

  if (state.editingRecipeId) {
    const idx = state.recipes.findIndex(r => r.id === state.editingRecipeId);
    if (idx > -1) state.recipes[idx] = recipeData;
  } else {
    state.recipes.push(recipeData);
  }
  state.editingRecipeId = undefined;
  state.recipeDraft = null;
  saveRecipes();
  toast('✅ Rezept gespeichert!');
  render();
}

/* ---------------- Wochenplan ---------------- */

function renderPlannerView() {
  const today = getTodayISO();
  const start = state.plannerWeekStart || mondayOf(today);
  const week = WEEKDAYS.map((day, i) => ({ day, date: addDays(start, i), iso: addDays(start, i) }));
  const planMap = {};
  state.plan.forEach(p => { if (!planMap[p.date]) planMap[p.date] = []; planMap[p.date].push(p); });

  let totalCost = 0;
  const daysWithPlans = week.filter(w => planMap[w.iso]);
  daysWithPlans.forEach(w => {
    (planMap[w.iso] || []).forEach(p => {
      const recipe = state.recipes.find(r => r.id === p.recipeId);
      if (recipe) {
        totalCost += recipe.ingredients.reduce((sum, ing) => sum + (ing.price || 0), 0);
      }
    });
  });

  const budget = state.settings.weeklyBudget;
  let budgetHtml = '';
  if (budget) {
    const remaining = budget - totalCost;
    const cls = remaining >= 0 ? 'budget-ok' : 'budget-over';
    budgetHtml = `<div class="budget-banner ${cls}">💰 Budget: ${budget.toFixed(2)} € | Verplant: ${totalCost.toFixed(2)} € | ${remaining >= 0 ? '✅ Noch ' + remaining.toFixed(2) + ' € übrig' : '⚠️ ' + Math.abs(remaining).toFixed(2) + ' € über Budget'}</div>`;
  }

  return `
    <div class="week-nav">
      <button class="btn-secondary btn-small" onclick="changeWeek(-1)">◀ Vorherige</button>
      <span style="font-weight:600;">${fmtDE(start)} – ${fmtDE(addDays(start,6))}</span>
      <button class="btn-secondary btn-small" onclick="changeWeek(1)">Nächste ▶</button>
      <button class="btn-secondary btn-small" onclick="state.plannerWeekStart=mondayOf(getTodayISO()); render();">Heute</button>
    </div>
    ${budgetHtml}
    <div class="week-days">
      ${week.map(w => {
        const items = planMap[w.iso] || [];
        return `
          <div class="day-card">
            <h3>${w.day} (${fmtDE(w.iso)}) <span style="font-size:.75rem;font-weight:400;color:var(--text-muted);">${items.length} Rezepte</span></h3>
            ${items.map(p => {
              const recipe = state.recipes.find(r => r.id === p.recipeId);
              if (!recipe) return `<div class="plan-item warn">⚠️ Rezept nicht gefunden</div>`;
              const cost = recipe.ingredients.reduce((sum, ing) => sum + (ing.price || 0), 0);
              return `<div class="plan-item"><span>${escapeHtml(recipe.name)} <span style="font-size:.75rem;color:var(--text-muted);">${cost.toFixed(2)} €</span></span>
                <button class="remove-x" onclick="removeFromPlan('${w.iso}','${p.recipeId}')">✕</button>
              </div>`;
            }).join('')}
            <div style="margin-top:8px;display:flex;gap:6px;flex-wrap:wrap;">
              <select id="add-recipe-${w.iso}" style="width:auto;font-size:.85rem;padding:6px 8px;flex:1;min-width:120px;">
                <option value="">+ Rezept hinzufügen</option>
                ${state.recipes.map(r => `<option value="${r.id}">${escapeHtml(r.name)}</option>`).join('')}
              </select>
              <button class="btn-secondary btn-small" onclick="addToPlan('${w.iso}')">Hinzufügen</button>
            </div>
          </div>
        `;
      }).join('')}
    </div>
  `;
}

function changeWeek(delta) {
  const current = state.plannerWeekStart || mondayOf(getTodayISO());
  state.plannerWeekStart = addDays(current, delta * 7);
  render();
}

function addToPlan(date) {
  const select = document.getElementById(`add-recipe-${date}`);
  if (!select) return;
  const recipeId = select.value;
  if (!recipeId) { toast('Bitte ein Rezept auswählen'); return; }
  const recipe = state.recipes.find(r => r.id === recipeId);
  if (!recipe) { toast('Rezept nicht gefunden'); return; }
  if (state.plan.some(p => p.date === date && p.recipeId === recipeId)) {
    toast('⚠️ Rezept ist bereits für diesen Tag geplant');
    return;
  }
  state.plan.push({ date, recipeId });
  savePlan();
  toast(`✅ "${recipe.name}" zum Plan hinzugefügt`);
  render();
}

function removeFromPlan(date, recipeId) {
  state.plan = state.plan.filter(p => !(p.date === date && p.recipeId === recipeId));
  savePlan();
  render();
}

function renderShoppingView() {
  const planMap = {};
  state.plan.forEach(p => {
    if (!planMap[p.date]) planMap[p.date] = [];
    planMap[p.date].push(p.recipeId);
  });

  const shoppingList = {};
  let totalCost = 0;

  Object.entries(planMap).forEach(([date, recipeIds]) => {
    recipeIds.forEach(recipeId => {
      const recipe = state.recipes.find(r => r.id === recipeId);
      if (!recipe) return;
      recipe.ingredients.forEach(ing => {
        const key = ing.name.toLowerCase();
        if (!shoppingList[key]) {
          shoppingList[key] = {
            name: ing.name,
            amount: 0,
            unit: ing.unit || 'Stück',
            category: ing.category || 'other',
            price: ing.price || 0,
            checked: state.settings.checked && state.settings.checked[key] || false,
            fromRecipes: []
          };
        }
        if (ing.amount !== null && ing.amount !== undefined) {
          shoppingList[key].amount += ing.amount;
        }
        if (!shoppingList[key].fromRecipes.includes(recipe.name)) {
          shoppingList[key].fromRecipes.push(recipe.name);
        }
        if (ing.price) totalCost += ing.price;
      });
    });
  });

  const list = Object.values(shoppingList);
  const checkedCount = list.filter(item => item.checked).length;

  const categoryOrder = state.settings.categoryOrder || CATEGORIES;
  list.sort((a, b) => {
    const idxA = categoryOrder.indexOf(a.category);
    const idxB = categoryOrder.indexOf(b.category);
    return (idxA === -1 ? 999 : idxA) - (idxB === -1 ? 999 : idxB);
  });

  const grouped = {};
  list.forEach(item => {
    if (!grouped[item.category]) grouped[item.category] = [];
    grouped[item.category].push(item);
  });

  return `
    <div class="toolbar">
      <span style="font-weight:600;">🛒 Einkaufsliste</span>
      <span class="item-meta">${list.length} Artikel (${checkedCount} erledigt)</span>
      <button class="btn-secondary btn-small" onclick="clearChecked()">✅ Erledigte entfernen</button>
      <button class="btn-secondary btn-small" onclick="toggleAllChecked()">☑️ Alle umschalten</button>
    </div>
    <div class="card">
      ${list.length === 0 ? `<div class="empty-state">📭 Keine Zutaten im Plan. Füge Rezepte zum Wochenplan hinzu.</div>` : `
        ${Object.entries(grouped).map(([category, items]) => `
          <div style="margin-bottom:14px;">
            <div style="font-weight:600;color:var(--primary-dark);margin-bottom:6px;font-size:.9rem;">${CATEGORY_LABELS[category] || category}</div>
            ${items.map(item => `
              <div class="shopping-item ${item.checked ? 'checked' : ''}">
                <input type="checkbox" ${item.checked ? 'checked' : ''} onchange="toggleShoppingItem('${escapeAttr(item.name)}')">
                <label style="flex:1;cursor:pointer;">${escapeHtml(item.name)}</label>
                <span class="item-meta">${item.amount ? item.amount.toFixed(1) : ''} ${item.unit}</span>
                ${item.price ? `<span class="item-meta">${item.price.toFixed(2)} €</span>` : ''}
                <span style="font-size:.7rem;color:var(--text-muted);max-width:120px;overflow:hidden;text-overflow:ellipsis;white-space:nowrap;">
                  ${item.fromRecipes.slice(0,2).join(', ')}${item.fromRecipes.length > 2 ? ` +${item.fromRecipes.length-2}` : ''}
                </span>
              </div>
            `).join('')}
          </div>
        `).join('')}
        ${totalCost > 0 ? `<div style="margin-top:12px;padding-top:10px;border-top:2px solid var(--border);font-weight:600;">💰 Geschätzter Gesamtpreis: ${totalCost.toFixed(2)} €</div>` : ''}
      `}
    </div>
  `;
}

function toggleShoppingItem(name) {
  if (!state.settings.checked) state.settings.checked = {};
  state.settings.checked[name.toLowerCase()] = !state.settings.checked[name.toLowerCase()];
  saveSettings();
  render();
}

function clearChecked() {
  const toRemove = Object.keys(state.settings.checked || {}).filter(k => state.settings.checked[k]);
  toRemove.forEach(k => delete state.settings.checked[k]);
  saveSettings();
  render();
  toast('✅ Erledigte Artikel entfernt');
}

function toggleAllChecked() {
  const planMap = {};
  state.plan.forEach(p => {
    if (!planMap[p.date]) planMap[p.date] = [];
    planMap[p.date].push(p.recipeId);
  });
  const allItems = new Set();
  Object.values(planMap).forEach(recipeIds => {
    recipeIds.forEach(recipeId => {
      const recipe = state.recipes.find(r => r.id === recipeId);
      if (recipe) {
        recipe.ingredients.forEach(ing => allItems.add(ing.name.toLowerCase()));
      }
    });
  });
  if (allItems.size === 0) { toast('Keine Artikel in der Liste'); return; }
  const allChecked = Array.from(allItems).every(name => state.settings.checked && state.settings.checked[name]);
  if (!state.settings.checked) state.settings.checked = {};
  allItems.forEach(name => { state.settings.checked[name] = !allChecked; });
  saveSettings();
  render();
  toast(allChecked ? '☑️ Alle abgehakt' : '☑️ Alle als erledigt markiert');
}

let dragCategory = null;
function dragStart(e) {
  dragCategory = e.target.dataset.category;
  e.dataTransfer.effectAllowed = 'move';
}
function dragOver(e) {
  e.preventDefault();
  e.dataTransfer.dropEffect = 'move';
}
function dropCategory(e) {
  e.preventDefault();
  const target = e.target.dataset.category;
  if (!dragCategory || !target || dragCategory === target) return;
  const order = state.settings.categoryOrder || [...CATEGORIES];
  const fromIdx = order.indexOf(dragCategory);
  const toIdx = order.indexOf(target);
  if (fromIdx === -1 || toIdx === -1) return;
  order.splice(toIdx, 0, order.splice(fromIdx, 1)[0]);
  state.settings.categoryOrder = order;
  saveSettings();
  render();
}

function saveApiKey(key) {
  state.settings.apiKey = key;
  saveSettings();
  toast('✅ API-Key gespeichert');
}

function saveAppliances() {
  const input = document.getElementById('custom-appliances');
  const items = input.value.split(',').map(s => s.trim()).filter(Boolean);
  state.settings.customAppliances = items;
  saveSettings();
  toast('✅ Geräte gespeichert');
  render();
}

function exportData() {
  const data = {
    recipes: state.recipes,
    plan: state.plan,
    settings: state.settings,
    exportDate: new Date().toISOString()
  };
  const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `herbi-data-${fmtISO(new Date())}.json`;
  a.click();
  URL.revokeObjectURL(url);
  toast('📤 Daten exportiert');
}

function importData(event) {
  const file = event.target.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.onload = function(e) {
    try {
      const data = JSON.parse(e.target.result);
      if (data.recipes) state.recipes = data.recipes;
      if (data.plan) state.plan = data.plan;
      if (data.settings) state.settings = Object.assign(state.settings, data.settings);
      saveRecipes();
      savePlan();
      saveSettings();
      render();
      toast('✅ Daten importiert!');
    } catch (err) {
      toast('❌ Fehler beim Import: ' + err.message);
    }
  };
  reader.readAsText(file);
  event.target.value = '';
}
